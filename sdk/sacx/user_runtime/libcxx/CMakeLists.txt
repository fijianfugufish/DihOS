cmake_minimum_required(VERSION 3.20)
project(DihOSLibcxxHeaderConfiguration NONE)
# A header-only target configuration, not an upstream libc++.a build.
if(NOT EXISTS "${LIBCXX_INCLUDE}/__config_site.in")
  message(FATAL_ERROR "Supply the pinned libc++ include tree")
endif()
set(_LIBCPP_ABI_VERSION 1)
set(_LIBCPP_ABI_NAMESPACE __1)
set(_LIBCPP_ABI_FORCE_ITANIUM 1)
set(_LIBCPP_DISABLE_VISIBILITY_ANNOTATIONS 1)
set(_LIBCPP_NO_VCRUNTIME 1)
set(_LIBCPP_HAS_UNICODE 1)
set(_LIBCPP_TYPEINFO_COMPARISON_IMPLEMENTATION 1)
set(_LIBCPP_PSTL_BACKEND_SERIAL 1)
set(_LIBCPP_HARDENING_MODE_DEFAULT 16)
set(_LIBCPP_ASSERTION_SEMANTIC_DEFAULT 32)
option(DIHOS_SYNCHRONIZATION "Build selected SACX-backed standard synchronization APIs" OFF)
if(DIHOS_SYNCHRONIZATION)
  set(_LIBCPP_HAS_THREADS 1)
  set(_LIBCPP_HAS_THREAD_API_EXTERNAL 1)
  set(_LIBCPP_HAS_MONOTONIC_CLOCK 1)
  configure_file(__external_threading "${CMAKE_BINARY_DIR}/__external_threading" COPYONLY)
endif()
# Other cmakedefine01 features remain zero: no filesystem/locale,
# random device, wide characters, terminal or timezone database in this port.
configure_file("${LIBCXX_INCLUDE}/__config_site.in" "${CMAKE_BINARY_DIR}/__config_site" @ONLY)
configure_file(assertion_handler.h "${CMAKE_BINARY_DIR}/__assertion_handler" COPYONLY)
